// 1. Configure Navigation Bar Title & Subtitle
await goat.navigation.setTitle({
  title: "Goat Mini App",
  subtitle: "High-Performance Engine"
});

// 2. Set Theme Color & Progress Indicator

  backgroundColor: "#1e293b",
  textColor: "#ffffff",
  isDark: true
});

  visible: true,
  percent: 65
});
